function flush(){

  if(e.error||e.afterError){errors.push(e);continue;}
  if(!['Entity.ApplyModifier','BlockDamage._AssignDynamicVar'].includes(e.name))continue;
  let source,amount,metric;
  if(e.name==='BlockDamage._AssignDynamicVar'){
   source=entity(ptr(e.currentBuff.source));amount=Number(BigInt(e.damageArg))/4294967296;metric='shield';
  }else{
   if(e.afterModifier.targetType!==0)continue;

  }
